cmake_minimum_required(VERSION 3.20)
project(ArmadaAnalyzerNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(ArmadaCompute SHARED Native/montecarlo.cpp)
add_library(AutoScopeMetadata SHARED Native/tds_metadata.cpp)
add_library(ArmadaSqlite SHARED Vendor/sqlite/sqlite3.c)
target_compile_definitions(ArmadaSqlite PRIVATE "SQLITE_API=__declspec(dllexport)" SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
if(MSVC)
  target_compile_options(ArmadaCompute PRIVATE /O2 /EHsc)
  target_compile_options(AutoScopeMetadata PRIVATE /O2 /EHsc)
  target_compile_options(ArmadaSqlite PRIVATE /O2)
else()
  target_compile_options(ArmadaCompute PRIVATE -O3)
  target_compile_options(AutoScopeMetadata PRIVATE -O3)
  target_compile_options(ArmadaSqlite PRIVATE -O3)
endif()
install(TARGETS ArmadaCompute ArmadaSqlite AutoScopeMetadata RUNTIME DESTINATION . LIBRARY DESTINATION .)
